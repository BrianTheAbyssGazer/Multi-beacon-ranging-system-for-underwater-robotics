void PingOut::set(uint16_t out_idx) {
    if (debug) {HAL_GPIO_WritePin(GPIOC, GPIO_PIN_10, GPIO_PIN_SET);}
    uint16_t Nbits_data=N_CYCLE*3;
    uint16_t i=out_idx;
    uint16_t end_idx;
    if(scheduled_idx<OUT_BUF_LEN/2){
    	scheduled_idx=OUT_BUF_LEN/2;
    	end_idx = OUT_BUF_LEN/2;
    }
    else {
    	scheduled_idx=0;
    	end_idx = OUT_BUF_LEN;
    }
    if (phase){
    	symbol_idx=data_idx/Nbits_data;
		gold_code_idx=(data_idx%Nbits_data)/N_CYCLE;
		if(codeword[data_idx] != goldcode[gold_code_idx]){
			out_buf[i]=BSRR_PC6_SET_MASK;
			i+=1;
	    	data_idx+=2;
		}
    }
    while (i<end_idx-1){
    	symbol_idx=data_idx/Nbits_data;
    	if(symbol_idx<5){
        	gold_code_idx=(data_idx%Nbits_data)/N_CYCLE;
        	if(codeword[data_idx] != goldcode[gold_code_idx]){
        		out_buf[i+1]=BSRR_PC6_SET_MASK;
        	}
        	else{
        		out_buf[i]=BSRR_PC6_SET_MASK;
        	}
        	i+=2;
        	data_idx+=2;
    	}
    	else {
    		sending=false;
    		break;
    	}
    }
    if(i!=end_idx){
    	symbol_idx=data_idx/Nbits_data;
    	if(symbol_idx<5){
        	gold_code_idx=(data_idx%Nbits_data)/N_CYCLE;
        	if(!(codeword[data_idx] != goldcode[gold_code_idx])){
        		out_buf[i]=BSRR_PC6_SET_MASK;
        	}
        	phase=true;
    	}
    	else sending=false;
    }
	clear_range=i;
    clear_idx = out_idx;
    time_to_clear =0;
    if (debug) {HAL_GPIO_WritePin(GPIOC, GPIO_PIN_10, GPIO_PIN_RESET);}
}
